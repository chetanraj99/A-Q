def equal_sum_partition(arr,n,w):
    t=[[True  if i == 0 else False if  j==0  else False for i in range(w+1)] for j in range(n+1)]
                
    for i in range(1,n+1):
        for j in range(1,w+1):
            if arr[i-1]<=j:
                t[i][j]=t[i-1][j-arr[i-1]] or t[i-1][j]
            else:
                t[i][j]=t[i-1][j]
    return t[n][w]

arr=[1,5,11,5]
s=sum(arr)
if s%2!=0:
    print(False)
elif s%2==0:
    print(equal_sum_partition(arr,len(arr),s))
