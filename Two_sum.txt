def knapSack(target, coins, n,l):

	if n < 0:
		return 0
	if target==0:
	    l.sort()
	    print(l)
	    return 1

	if (coins[n-1] > target):
		return knapSack(target, coins, n-1,l)

	else:
	    l.append(n-1)
	    return max(knapSack(target-coins[n-1], coins, n-1,l),knapSack(target, coins, n-1,l))
	   


coins = [ 10,30,20]
target = 50
n = len(coins)
print(knapSack(target, coins, n,[]))
