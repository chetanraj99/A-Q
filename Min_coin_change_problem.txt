
#include <iostream>
using namespace std;
#include <bits/stdc++.h> 
int Min_Coin_change_problem(int n, int k, vector<int> &arr) {
    
    int t[n+1][k+1];
    for(int i=0;i<n+1;i++){
        for(int j=0;j<k+1;j++){
            if(i==0)
                t[i][j]=1e9;
            else
                t[i][j]=0;
        }
    }
    
    
    
    for(int i=1;i<n+1;i++){
        for(int j=1;j<k+1;j++){
            if(arr[i-1]<=j)
                t[i][j]=min(t[i][j-arr[i-1]] +1, t[i-1][j]);
            else
                t[i][j]=t[i-1][j];
        }
    }
    
    for(int i=0;i<n+1;i++){
        for(int j=0;j<k+1;j++){
            cout<<t[i][j]<<" ";
        }
        cout<<endl;
    }
    
      return t[n][k];
    
 }
