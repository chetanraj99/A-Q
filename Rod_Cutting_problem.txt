def rod_cutting(lenth,price,n,w):
    t=[[-1 for i in range(w+1)] for j in range(n+1)]
    for i in range(n+1):
        for j in range(w+1):
            if i==0 or j==0:
                t[i][j]=0
    
    for i in range(1,n+1):
        for j in range(1,w+1):
            if lenth[i-1]<=j:
                t[i][j]=max(price[i-1]+t[i][j-lenth[i-1]],t[i-1][j])
            else:
                t[i][j]=t[i-1][j]
    return t[n][w]

print(rod_cutting([1,2,3,4],[5,6,8,8],4,4))
