def canSum_rec(wt,w,n,memo={}):
    print(memo)
    if n in memo:
        return memo[n]
    if n==0 and w!=0:
        return False
    if w==0:
        return True
    if wt[n-1]<=w:
        memo[n]= (canSum_rec(wt,w-wt[n-1],n-1,memo) or canSum_rec(wt,w,n-1,memo))
        return memo[n]
    elif wt[n-1]>w:
        memo[n]= canSum_rec(wt,w,n-1)
        return memo[n]
    
print(canSum_rec([25,654,1000,125],150,4))
