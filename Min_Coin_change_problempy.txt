#Minimum number of coins to make


def Coin_change(coins,n,w):
    t=[[float('inf')  if i == 0 else 0 if  j==0  else 0 for j in range(w+1)] for i in range(n+1)]
                
    for i in range(1,n+1):
        for j in range(1,w+1):
            if coins[i-1]<=j:
                t[i][j]=min(1+t[i][j-coins[i-1]] , t[i-1][j])
            else:
                t[i][j]=t[i-1][j]
    print(t)
    return t[n][w]
