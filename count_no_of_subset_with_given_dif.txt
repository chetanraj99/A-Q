def wayofsum(arr,n,w):
    t=[[1  if i == 0 else 0 if  j==0  else 0 for i in range(w+1)] for j in range(n+1)]
                
    for i in range(1,n+1):
        for j in range(1,w+1):
            if arr[i-1]<=j:
                t[i][j]=t[i-1][j-arr[i-1]] + t[i-1][j]
            else:
                t[i][j]=t[i-1][j]
    return t[n][w]
arr=[1,1,2,3]
diff=1
s=sum(arr)
target=(diff+s)//2
print(wayofsum(arr,len(arr),target))
