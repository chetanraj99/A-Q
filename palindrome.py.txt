def Partition_palindrome(s,i,j):
    if i>=j:
        return 0
    if s[i:j+1]==s[i:j+1][::-1] :
        return 0
    mn=float('inf')
    for k in range(i,j):
        temp=Partition_palindrome(s,i,k)+Partition_palindrome(s,k+1,j)+1
        if temp<mn:
            mn=temp
    return mn
print(Partition_palindrome('ababa',0,4))
print(Partition_palindrome('abbac',0,4))
print(Partition_palindrome('abcde',0,4))
print(Partition_palindrome('geek',0,4))
