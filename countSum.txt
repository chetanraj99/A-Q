def cansum(arr,j,i):
    n=i
    w=j
    table=[[1  if i == 0 else 0 if  j==0  else 0 for i in range(w+1)] for j in range(n+1)]
    for i in range(n+1):
        for j in range(1,w+1):
            if arr[i-1]<=j:
                table[i][j]=table[i][j-arr[i-1]] + table[i-1][j]
            else:
                table[i][j]=table[i-1][j]
    print(table)
    return table[i][j]
        
print(cansum([2,3,5],8,3))
