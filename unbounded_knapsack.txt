def unbounded_knapsack(wt,val,n,w):
    t=[[-1 for i in range(w+1)] for j in range(n+1)]
    for i in range(n+1):
        for j in range(w+1):
            if i==0 or j==0:
                t[i][j]=0
    
    for i in range(1,n+1):
        for j in range(1,w+1):
            if wt[i-1]<=j:
                t[i][j]=max(val[i-1]+t[i][j-wt[i-1]],t[i-1][j])
            else:
                t[i][j]=t[i-1][j]
    return t[n][w]

print(unbounded_knapsack([1,3,4,5],[1,4,5,7],4,7))

def unbounded_knapsack(wt,val,w,n):
    if n==0 or w==0:
        return 0
    if wt[n-1]<=w:
        return max(val[n-1]+unbounded_knapsack(wt,val,w-wt[n-1],n),unbounded_knapsack(wt,val,w,n-1))
    elif wt[n-1]>w:
        return unbounded_knapsack(wt,val,w,n-1)
        
print(unbounded_knapsack([1,3,4,5],[1,4,5,7],7,4))
