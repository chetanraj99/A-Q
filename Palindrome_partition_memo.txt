#Palindrome Partitioning Memoization
def Palindrome(s,i,j):
    if t[i][j]!=0:
        return t[i][j]
    if i>=j:
        return 0
    if s==s[::-1]:
        return 0
    mn=float('inf')
    for k in range(i,j):
        temp=Palindrome(s,i,k)+Palindrome(s,k+1,j)+1
        t[i][j]=min(temp,mn)
    return t[i][j]
t=[[-1 for i in range(3+1)] for j in range(3+1)]
print(Palindrome('geek',1,3))
