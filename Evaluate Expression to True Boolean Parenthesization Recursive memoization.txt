def solve(s,i,j,isTrue,memo={}):
    key=f'{i}{j}{isTrue}'
    if i>j:
        return 0
    if i==j:
        if isTrue==True:
            return 1 if s[i]=='T' else 0
        else:
            return 1 if s[i]=='F' else 0
    
    ans=0
    for k in range(i+1,j,2):
        LT=solve(s,i,k-1,True,memo)
        LF=solve(s,i,k-1,False,memo)
        RT=solve(s,k+1,j,True,memo)
        RF=solve(s,k+1,j,False,memo)
        
        if s[k]=='&':
            if isTrue==True:
                ans=ans+LT*RT
            else:
                ans+=LF*RT+LF*RF+LT*RF
        elif s[k]=='|':
            if isTrue==True:
                ans+=LT*RT+LT*RF+LF*RT
            else:
                ans+=LF*RF
        elif s[k]=='^':
            if isTrue==True:
                ans+=LF*RT+LT*RF
            else:
                ans+=LT*RT+LF*RF
    memo[key]=ans
    return ans

print(solve("T|T&F^T",0,6,True))
