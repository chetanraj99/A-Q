def wayofsum(arr,n,w):
    t=[[1  if i == 0 else 0 if  j==0  else 0 for i in range(w+1)] for j in range(n+1)]
                
    for i in range(1,n+1):
        for j in range(1,w+1):
            if arr[i-1]<=j:
                t[i][j]=t[i][j-arr[i-1]] + t[i-1][j]
            else:
                t[i][j]=t[i-1][j]
    return t[n][w]
print(wayofsum([2,3,5],3,5))
