def gridtraveler(m,n):
    table=[[0 for i in range(n+1)] for y in range(m+1)]
    table[1][1]=1
    for i in range(m+1):
        for j in range(n+1):
            try:
                current=table[i][j]
                if j+1<=n:
                    table[i][j+1]+=current
                if i+1<=m:
                    table[i+1][j]+=current
            except:
                continue
    return table[m][n]

print(gridtraveler(1,1))
print(gridtraveler(2,3))
print(gridtraveler(3,2))
print(gridtraveler(3,3))
print(gridtraveler(18,18))
