#Sequence pattern matching
def lps(x,y,m,n):
    t=[[0 for j in range(n+1)] for i in range(m+1)]
    for i in range(1,m+1):
        for j in range(1,n+1):
            if x[i-1]==y[j-1]:
                t[i][j]= 1+t[i-1][j-1]
            else:
                t[i][j]= max(t[i][j-1],t[i-1][j])
    return t[m][n]
        
s1='axz'
s2='abcdxez'
print(len(s1)==lps(s1,s2,len(s1),len(s2)))
